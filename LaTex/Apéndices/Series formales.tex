Damos aquí las definiciones y los resultados de los que nos servimos a lo largo de todo el trabajo. Para un desarrollo completo de esta teoría se puede consultar \cite{sambale}.

\begin{definicion}
    Sea $\mathcal{R}$ un dominio íntegro. Se llama \textbf{anillo de series formales en la variable $T$ con coeficientes en $\mathcal{R}$} al conjunto de sucesiones infinitas en $\mathcal{R}$, es decir, de elementos
    \begin{equation*}
        \alpha = \{a_0, a_1, a_2, \dots \} \hspace{0.25cm} a_n \in \mathcal{R}, \forall n \in \N 
    \end{equation*}
    dotado de las operaciones de suma y producto dadas por
    \begin{equation*}
    \alpha + \beta \equiv \{a_0 + b_0, a_1 + b_1, a_2 + b_2, \dots \}
    \end{equation*}
    \begin{equation*}
    \alpha \cdot \beta \equiv \left\{a_0  b_0, a_0b_1 + a_1b_0, a_0b_2 + a_1b_1 + a_2b_0, \dots, \sum_{k=0}^na_kb_{n-k}, \dots \right\} 
    \end{equation*}
    y donde se define \textbf{la variable formal} $T$ como el elemento $T \equiv \delta_{n,1} = (0,1,0, \dots)$. Se simboliza por $\mathcal{R} \llbracket T \rrbracket$ o por $\F$, si no hay riesgo de confusión, a este conjunto y a sus elementos se les llaman \textbf{series formales en la variable T}.
\end{definicion}

\begin{teorema}
    $(\F, +, \cdot)$ es un anillo. El elemento neutro y la unidad son, respectivamente, $0 \equiv \{0,0,0,\dots\} \text{ y } 1\equiv \{1, 0, 0, \dots\}$ y el elemento opuesto o inverso respecto a la suma es $ -\alpha \equiv \{-a_0, -a_1, -a_2, \dots \}$
\end{teorema}

\begin{definicion} 
    Dada $\alpha = (a_0, a_1, \dots) \in \F$, se define el \textbf{orden de $\alpha$} como
    \begin{equation*}
        \ord(\alpha) = \min \{ n \geq 0 : a_n \neq 0 \}
    \end{equation*}
    En particular, se define $\ord(0) = \infty$
\end{definicion}

\begin{proposicion} El orden es una valoración discreta, es decir, verifica las siguientes propiedades.
    \begin{itemize}
        \item $\ord(\alpha) = \infty \iff \alpha = 0$
        \item $\ord(\alpha + \beta) \geq \min(\ord(\alpha), \ord(\beta))$
        \item $\ord(\alpha \cdot \beta) \geq \ord(\alpha) + \ord(\beta)$ y con igualdad si $\mathcal{R}$ es un dominio.
    \end{itemize}
\end{proposicion}

\begin{teorema}
    Un elemento $\alpha = \{a_0, a_1, \dots\} \in \F$ tiene inverso multiplicativo si y solo si $a_0 \in \mathcal{R}^*$. En particular, si $\mathcal{R}$ es un cuerpo, $\alpha$ tiene inverso multiplicativo si y solo si $\ord(\alpha) \neq 0$ 
\end{teorema}

\begin{proposicion}
    Dado un entero positivo $n$ y $\alpha \in \F^*$, se tiene que $(\alpha^{-1})^n = (\alpha^n)^{-1}$, siendo $\alpha^{-1}$ el inverso multiplicativo de $\alpha$.
\end{proposicion}

\begin{definicion}
    Se define el \textbf{valor absoluto formal} como la aplicación
    \begin{align*}
        | \quad |_f \colon & \F \longrightarrow \mathcal{R} \\
        & \alpha \mapsto |\alpha|_f = \begin{dcases}
            e^{-\ord(\alpha)} & \alpha \neq 0\\
            0 & \alpha = 0
        \end{dcases}
    \end{align*}
    En particular, este define la métrica $\text{d}(\alpha, \beta) = |\alpha - \beta|_f$. Se llama \textbf{topología formal} a la topología de $\F$ asociada a esta métrica.
\end{definicion}

\begin{teorema}
    $\F$ es completo con la topología formal.
\end{teorema}

\begin{teorema}
    El anillo de polinomios es denso sobre $\F$. En consecuencia, para toda serie formal $\alpha = \{ a_0, a_1, a_2,\dots\}$ se tiene que
    \begin{equation*}
        \alpha = \sum_{k=0}^{\infty} a_k T^k = \lim_{N \longrightarrow \infty} \left(\sum_{k=0}^{N} a_k T^k\right)
    \end{equation*}
\end{teorema}

Este resultado prueba así que se pueden intepretar a los elementos de $\F$ como series infinitas en la variable $T$. De esta manera, cada $\alpha = \{ a_0, a_1, a_2,\dots\}$ se puede pensar como $\alpha \equiv F(T) = \sum_{k=0}^\infty a_k T^k$

\begin{teorema}
    Dada una sucesión de series formales $F_n(T)$, la serie de series formales $\sum_{n=0}^\infty F_n(T)$ es convergente si y solo si el límite de $F_n(T)$ es 0.
\end{teorema}

En lo que sigue, supondremos que $\mathcal{R} = \field$ es un cuerpo de característica cero.

\begin{teorema}
    $\F$ es un anillo local con ideal maximal $\maximal = (T)$. Además, sus únicos ideales no nulos son las potencias $\maximal^n$.
\end{teorema}

\begin{corolario}
    $\F$ es un anillo de valoración discreta.
\end{corolario}

\begin{definicion}
    Sea $F \in \F$ y $G \in \maximal$. Se define la \textbf{composición de $F$ con $G$} como la serie formal
    \begin{equation*}
        F \circ G \equiv F(G) = \sum_{k=0}^\infty a_k G^n \text{ siendo } F(T) = \sum_{k=0}^\infty a_k T^k
    \end{equation*}
\end{definicion}

\begin{teorema}
    Dada $F \in \F$, esta admite una inversa respecto a la composición si y solo si $F \in \maximal$ pero $F \not \in \maximal^2$
\end{teorema}

\begin{proposicion}
    La composición de series formales verifica

    \begin{tabular}{@{}ll@{}}
    $\bullet$ $(F + G) \circ H = F \circ H + G \circ H$ & 
    \quad $\bullet$ $F \circ T = T \circ F = F$ \\[6pt]
    $\bullet$ $(F \cdot G) \circ H = (F \circ H) \cdot (G \circ H)$ & 
    \quad $\bullet$ $(F \circ G) \circ H = F \circ (G \circ H)$ \\[6pt]
    $\bullet$ $F, G \in \maximal \implies F \circ G \in \maximal$ & \\
    \end{tabular}
\end{proposicion}

\begin{definicion}
    Se dice que la sucesión de series formales $F_n(T)$ forman una \textbf{pseudobase de $\F$} si para toda serie formal $F(T) \in \F$ existen constantes únicas $a_n$ tales que $F(T) = \sum_{n=0}^\infty a_n F_n(T)$
\end{definicion}

\begin{proposicion}
    Si $\ord(F) = 1$, entonces $\{F^n(T)\}_{n \geq 0}$ es una pseudobase de $\F$.
\end{proposicion}

\begin{definicion}
    Se define la \textbf{derivada formal} como el operador
    \begin{align*}
        \partial \colon \F & \longrightarrow \F \\
        \sum_{k=0}^{\infty} a_k T^k & \mapsto \sum_{k=1}^{\infty} ka_k T^{k-1}
    \end{align*}
    En particular, se tienen las \textbf{derivadas sucesivas de orden $n$} $\partial^n$
    \begin{equation*}
        \partial^n F(T) \equiv \partial \circ \partial^{n-1} F(T) = \sum_{k=n}^{\infty} a_k (k)_n T^{k-n}
    \end{equation*}
\end{definicion}

\begin{proposicion}
    Si $\ord(F) = n$, entonces $\ord(\partial F) = \ord(F) - 1$, para todo $n \geq 1$.
\end{proposicion}

\begin{proposicion}
    La derivada formal verifica las siguientes propiedades
    \begin{itemize}
        \item $\partial$ es $\field$-lineal y $\partial \field = 0$
        \item $\partial(F \cdot G) = \partial(F) \cdot G + F (\partial G)$
        \item $\partial (\frac{F}{G}) = \frac{\partial(F) \cdot G - F (\partial G)}{G^2}$ si $G \in \F^*$ (donde $\frac{F}{G} = F \cdot \frac{1}{G}$).
        \item $\partial(F \circ G) = ((\partial F) \circ G) \cdot G$ si $G \in \maximal$.
    \end{itemize}
\end{proposicion}

\begin{proposicion}
    Dada $F(T) \in \F$, esta se puede escribir como
    \begin{equation*}
        F(T) = \sum_{k=0}^\infty \frac{(\partial^k F)(0)}{k!}T^k
    \end{equation*}
    donde $(\partial^k F)(0)$ simboliza al escalar que resulta de sustituir la variable formal $T$ por $0$ en $\partial^k F$ (entendiendo esto como una serie infinita en $\field$).
\end{proposicion}
